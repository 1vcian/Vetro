# these the oracle is the native run on an aarch64 Linux host (the CI linux
# job); elsewhere they are skipped. Every entry is verified by comparing native
# and QEMU in the same environment (as an unprivileged user).
# Format: name  # native outcome / QEMU outcome: reason

clone08          # native 3 TPASS / QEMU none: CLONE_PARENT and *_SETTID emulated differently

fcntl40_64       # like fcntl40
futex_wait03     # native TPASS / QEMU TBROK: thread state in /proc differs under QEMU

getcwd01         # native 5 TPASS / QEMU TFAIL: getcwd with an invalid buffer

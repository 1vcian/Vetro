# Test LTP su cui qemu-aarch64 (user mode) si discosta dal kernel Linux: per
# questi l'oracolo è l'esecuzione nativa su un host Linux aarch64 (job linux
# della CI); altrove sono saltati. Ogni voce è verificata confrontando nativo
# e QEMU nello stesso ambiente (da utente non privilegiato).
# Formato: nome  # esito nativo / esito QEMU: motivo
clone02          # nativo 2 TPASS / QEMU TFAIL: la clone emulata non si comporta come quella del kernel
clone05          # nativo TPASS / QEMU TFAIL: idem (CLONE_VFORK)
clone08          # nativo 3 TPASS / QEMU nessuno: CLONE_PARENT e *_SETTID emulati diversamente
fcntl40          # nativo 3 TPASS / QEMU TFAIL: F_CREATED_QUERY non implementato in QEMU
fcntl40_64       # come fcntl40
futex_wait03     # nativo TPASS / QEMU TBROK: stato dei thread in /proc diverso sotto QEMU
futex_waitv01    # nativo 9 TPASS / QEMU TCONF: futex_waitv (449) non implementata in QEMU
futex_waitv02    # come futex_waitv01
futex_waitv03    # come futex_waitv01
getcwd01         # nativo 5 TPASS / QEMU TFAIL: getcwd con buffer non valido
kill11           # nativo 24 TPASS / QEMU TCONF: l'harness avvia QEMU con `ulimit -c 0` (anche il limite hard)
mmap08           # nativo TPASS / QEMU TFAIL: errore di mmap su un descrittore non valido
mmap17           # nativo TPASS / QEMU TFAIL: MAP_FIXED_NOREPLACE non rispettato
mremap03         # nativo TPASS / QEMU TFAIL: mremap su un intervallo non mappato
waitpid01        # nativo 146 TPASS / QEMU TCONF: come kill11 (core dump)
